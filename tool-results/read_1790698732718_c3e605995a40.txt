     1→'use client'
     2→
     3→import { useEffect, useState, useCallback } from 'react'
     4→import { toast } from 'sonner'
     5→import { LogOut } from 'lucide-react'
     6→import AdminLogin from './login'
     7→import { API, getApiUrl, apiFetch } from '@/lib/api-config'
     8→
     9→// ─── Types ───
    10→interface ProductImage { id: string; url: string; alt?: string; sortOrder: number; isPrimary: boolean }
    11→interface Brand { id: string; name: string; slug: string }
    12→interface Category { id: string; name: string; slug: string; icon?: string; color?: string }
    13→interface Product {
    14→  id: string; name: string; slug: string; model: string; price: number
    15→  description?: string; size?: string; color: string; isActive: boolean; featured: boolean; sortOrder: number
    16→  images: ProductImage[]; brand: Brand; category: Category
    17→}
    18→
    19→type Tab = 'products' | 'add' | 'categories' | 'brands' | 'orders'
    20→
    21→// ─── Component ───
    22→export default function AdminDashboard() {
    23→  // ─── Auth state (must be before conditional returns) ───
    24→  const [authenticated, setAuthenticated] = useState(false)
    25→  const [checkingAuth, setCheckingAuth] = useState(true)
    26→
    27→  // ─── All state hooks declared first (React rules) ───
    28→  const [tab, setTab] = useState<Tab>('products')
    29→  const [products, setProducts] = useState<Product[]>([])
    30→  const [categories, setCategories] = useState<Category[]>([])
    31→  const [brands, setBrands] = useState<Brand[]>([])
    32→  const [orders, setOrders] = useState<any[]>([])
    33→  const [loading, setLoading] = useState(true)
    34→  const [searchQuery, setSearchQuery] = useState('')
    35→
    36→  // Edit state
    37→  const [editingProduct, setEditingProduct] = useState<Product | null>(null)
    38→
    39→  // Add product form
    40→  const [form, setForm] = useState({
    41→    name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827',
    42→    categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0,
    43→    imageUrls: ''  // comma-separated
    44→  })
    45→
    46→  // Add category form
    47→  const [catForm, setCatForm] = useState({ name: '', slug: '', icon: '', color: '#1a1a2e' })
    48→  // Add brand form
    49→  const [brandForm, setBrandForm] = useState({ name: '', slug: '', logoUrl: '' })
    50→
    51→  // ─── Load data ───
    52→  const loadData = useCallback(async () => {
    53→    try {
    54→      const [prodRes, catRes, brandRes] = await Promise.all([
    55→        apiFetch(`${API}/products?limit=500`),
    56→        apiFetch(`${API}/categories`),
    57→        apiFetch(`${API}/brands`),
    58→      ])
    59→      const prodData = await prodRes.json()
    60→      const catData = await catRes.json()
    61→      const brandData = await brandRes.json()
    62→      setProducts(prodData.products || [])
    63→      setCategories(catData || [])
    64→      setBrands(brandData || [])
    65→    } catch {
    66→      toast.error('Failed to load data')
    67→    } finally {
    68→      setLoading(false)
    69→    }
    70→  }, [])
    71→
    72→  const loadOrders = useCallback(async () => {
    73→    try {
    74→      const res = await apiFetch(`${API}/orders?limit=100`)
    75→      const data = await res.json()
    76→      setOrders(data.orders || [])
    77→    } catch {
    78→      toast.error('Failed to load orders')
    79→    }
    80→  }, [])
    81→
    82→  useEffect(() => { loadData() }, [loadData])
    83→  useEffect(() => { if (tab === 'orders') loadOrders() }, [tab, loadOrders])
    84→
    85→  // ─── Auth check ───
    86→  useEffect(() => {
    87→    // First check localStorage for client-side auth (works on GitHub Pages)
    88→    const clientAuth = localStorage.getItem('admin_auth')
    89→    const authTs = localStorage.getItem('admin_ts')
    90→    const EIGHT_HOURS = 8 * 60 * 60 * 1000
    91→    if (clientAuth && authTs && (Date.now() - Number(authTs)) < EIGHT_HOURS) {
    92→      setAuthenticated(true)
    93→      setCheckingAuth(false)
    94→      return
    95→    }
    96→    // Otherwise try server-side verification
    97→    apiFetch(getApiUrl('/api/admin/verify'))
    98→      .then(res => res.ok ? setAuthenticated(true) : setAuthenticated(false))
    99→      .catch(() => setAuthenticated(false))
   100→      .finally(() => setCheckingAuth(false))
   101→  }, [])
   102→
   103→  const handleLogout = async () => {
   104→    localStorage.removeItem('admin_auth')
   105→    localStorage.removeItem('admin_ts')
   106→    await apiFetch(getApiUrl('/api/admin/logout'), { method: 'POST' }).catch(() => {})
   107→    setAuthenticated(false)
   108→    toast.success('Logged out')
   109→  }
   110→
   111→  // ─── Product CRUD ───
   112→  const deleteProduct = async (id: string) => {
   113→    if (!confirm('Delete this product? This cannot be undone.')) return
   114→    try {
   115→      await apiFetch(`${API}/products/${id}`, { method: 'DELETE' })
   116→      toast.success('Product deleted')
   117→      loadData()
   118→    } catch {
   119→      toast.error('Delete failed')
   120→    }
   121→  }
   122→
   123→  const toggleActive = async (product: Product) => {
   124→    try {
   125→      await apiFetch(`${API}/products/${product.id}`, {
   126→        method: 'PUT',
   127→        headers: { 'Content-Type': 'application/json' },
   128→        body: JSON.stringify({ ...product, isActive: !product.isActive, brandId: product.brand.id, categoryId: product.category.id }),
   129→      })
   130→      toast.success(product.isActive ? 'Product hidden' : 'Product visible')
   131→      loadData()
   132→    } catch {
   133→      toast.error('Update failed')
   134→    }
   135→  }
   136→
   137→  const toggleFeatured = async (product: Product) => {
   138→    try {
   139→      await apiFetch(`${API}/products/${product.id}`, {
   140→        method: 'PUT',
   141→        headers: { 'Content-Type': 'application/json' },
   142→        body: JSON.stringify({ ...product, featured: !product.featured, brandId: product.brand.id, categoryId: product.category.id }),
   143→      })
   144→      toast.success(product.featured ? 'Unfeatured' : 'Featured!')
   145→      loadData()
   146→    } catch {
   147→      toast.error('Update failed')
   148→    }
   149→  }
   150→
   151→  // ─── Add product ───
   152→  const handleAddProduct = async (e: React.FormEvent) => {
   153→    e.preventDefault()
   154→    if (!form.name || !form.model || !form.categoryId || !form.brandId) {
   155→      toast.error('Fill in all required fields')
   156→      return
   157→    }
   158→    try {
   159→      const images = form.imageUrls.split(',').map(u => u.trim()).filter(Boolean)
   160→      const res = await apiFetch(`${API}/products`, {
   161→        method: 'POST',
   162→        headers: { 'Content-Type': 'application/json' },
   163→        body: JSON.stringify({ ...form, images, price: Number(form.price), sortOrder: Number(form.sortOrder) }),
   164→      })
   165→      if (!res.ok) throw new Error()
   166→      toast.success('Product added!')
   167→      setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' })
   168→      loadData()
   169→      setTab('products')
   170→    } catch {
   171→      toast.error('Failed to add product')
   172→    }
   173→  }
   174→
   175→  // ─── Update product ───
   176→  const handleUpdateProduct = async (e: React.FormEvent) => {
   177→    e.preventDefault()
   178→    if (!editingProduct) return
   179→    try {
   180→      const images = form.imageUrls.split(',').map(u => u.trim()).filter(Boolean)
   181→      await apiFetch(`${API}/products/${editingProduct.id}`, {
   182→        method: 'PUT',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({
   185→          name: form.name, model: form.model, price: Number(form.price),
   186→          description: form.description, size: form.size, color: form.color,
   187→          categoryId: form.categoryId, brandId: form.brandId,
   188→          featured: form.featured, isActive: form.isActive, sortOrder: Number(form.sortOrder),
   189→          images
   190→        }),
   191→      })
   192→      toast.success('Product updated!')
   193→      setEditingProduct(null)
   194→      setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' })
   195→      loadData()
   196→      setTab('products')
   197→    } catch {
   198→      toast.error('Failed to update product')
   199→    }
   200→  }
   201→
   202→  const startEdit = (p: Product) => {
   203→    setEditingProduct(p)
   204→    setForm({
   205→      name: p.name, model: p.model, price: p.price, description: p.description || '',
   206→      size: p.size || 'N/A', color: p.color, categoryId: p.category.id, brandId: p.brand.id,
   207→      featured: p.featured, isActive: p.isActive, sortOrder: p.sortOrder,
   208→      imageUrls: p.images.map(i => i.url).join(', ')
   209→    })
   210→    setTab('add')
   211→  }
   212→
   213→  // ─── Add category ───
   214→  const handleAddCategory = async (e: React.FormEvent) => {
   215→    e.preventDefault()
   216→    if (!catForm.name) return
   217→    try {
   218→      const slug = catForm.slug || catForm.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   219→      const res = await apiFetch(`${API}/categories`, {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify({ ...catForm, slug }),
   223→      })
   224→      if (!res.ok) throw new Error()
   225→      toast.success('Category added!')
   226→      setCatForm({ name: '', slug: '', icon: '', color: '#1a1a2e' })
   227→      loadData()
   228→    } catch {
   229→      toast.error('Failed to add category')
   230→    }
   231→  }
   232→
   233→  // ─── Add brand ───
   234→  const handleAddBrand = async (e: React.FormEvent) => {
   235→    e.preventDefault()
   236→    if (!brandForm.name) return
   237→    try {
   238→      const slug = brandForm.slug || brandForm.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   239→      const res = await apiFetch(`${API}/brands`, {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify({ ...brandForm, slug }),
   243→      })
   244→      if (!res.ok) throw new Error()
   245→      toast.success('Brand added!')
   246→      setBrandForm({ name: '', slug: '', logoUrl: '' })
   247→      loadData()
   248→    } catch {
   249→      toast.error('Failed to add brand')
   250→    }
   251→  }
   252→
   253→  // ─── Delete category ───
   254→  const deleteCategory = async (id: string) => {
   255→    if (!confirm('Delete this category? All products in it will also be deleted!')) return
   256→    try {
   257→      await apiFetch(`${API}/categories/${id}`, { method: 'DELETE' })
   258→      toast.success('Category deleted')
   259→      loadData()
   260→    } catch { toast.error('Delete failed') }
   261→  }
   262→
   263→  // ─── Delete brand ───
   264→  const deleteBrand = async (id: string) => {
   265→    if (!confirm('Delete this brand? All products under it will also be deleted!')) return
   266→    try {
   267→      await apiFetch(`${API}/brands/${id}`, { method: 'DELETE' })
   268→      toast.success('Brand deleted')
   269→      loadData()
   270→    } catch { toast.error('Delete failed') }
   271→  }
   272→
   273→  const fmtPrice = (p: number) => `$${p.toLocaleString('en-JM', { minimumFractionDigits: 2 })}`
   274→
   275→  const filtered = products.filter(p =>
   276→    !searchQuery || p.name.toLowerCase().includes(searchQuery.toLowerCase()) || p.model.toLowerCase().includes(searchQuery.toLowerCase())
   277→  )
   278→
   279→  // ─── Auth gate (after all hooks) ───
   280→  if (checkingAuth) {
   281→    return (
   282→      <div className="min-h-screen bg-slate-900 flex items-center justify-center">
   283→        <div className="flex items-center gap-3 text-slate-400">
   284→          <svg className="animate-spin h-6 w-6" viewBox="0 0 24 24">
   285→            <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" fill="none" />
   286→            <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" />
   287→          </svg>
   288→          Verifying session...
   289→        </div>
   290→      </div>
   291→    )
   292→  }
   293→
   294→  if (!authenticated) {
   295→    return <AdminLogin onLogin={() => setAuthenticated(true)} />
   296→  }
   297→
   298→  // ─── Render ───
   299→  return (
   300→    <div className="min-h-screen bg-gray-50">
   301→      {/* Header */}
   302→      <header className="bg-[#1a1a2e] text-white shadow-lg">
   303→        <div className="max-w-7xl mx-auto px-4 py-4 flex items-center justify-between">
   304→          <div>
   305→            <h1 className="text-xl font-bold">PixelWire Admin</h1>
   306→            <p className="text-xs opacity-60">Product &amp; Inventory Management</p>
   307→          </div>
   308→          <div className="flex items-center gap-3">
   309→            <a href="/" className="text-sm bg-white/10 hover:bg-white/20 px-3 py-1.5 rounded-lg transition">
   310→              View Storefront
   311→            </a>
   312→            <button
   313→              onClick={handleLogout}
   314→              className="text-sm bg-white/10 hover:bg-red-500/20 px-3 py-1.5 rounded-lg transition flex items-center gap-1.5"
   315→            >
   316→              <LogOut className="w-3.5 h-3.5" />
   317→              Logout
   318→            </button>
   319→          </div>
   320→        </div>
   321→      </header>
   322→
   323→      {/* Tabs */}
   324→      <nav className="bg-white border-b shadow-sm sticky top-0 z-40">
   325→        <div className="max-w-7xl mx-auto px-4 flex gap-1 overflow-x-auto">
   326→          {([
   327→            ['products', 'Products', `${products.length} items`],
   328→            ['add', editingProduct ? 'Edit Product' : 'Add Product', ''],
   329→            ['categories', 'Categories', `${categories.length}`],
   330→            ['brands', 'Brands', `${brands.length}`],
   331→            ['orders', 'Orders', ''],
   332→          ] as [Tab, string, string][]).map(([key, label, count]) => (
   333→            <button
   334→              key={key}
   335→              onClick={() => setTab(key)}
   336→              className={`px-4 py-3 text-sm font-medium border-b-2 transition whitespace-nowrap ${
   337→                tab === key
   338→                  ? 'border-[#1a1a2e] text-[#1a1a2e]'
   339→                  : 'border-transparent text-gray-500 hover:text-gray-700'
   340→              }`}
   341→            >
   342→              {label}{count ? ` (${count})` : ''}
   343→            </button>
   344→          ))}
   345→        </div>
   346→      </nav>
   347→
   348→      <main className="max-w-7xl mx-auto px-4 py-6">
   349→        {/* ─── PRODUCTS TAB ─── */}
   350→        {tab === 'products' && (
   351→          <div>
   352→            <div className="flex gap-3 items-center mb-4">
   353→              <input
   354→                type="text"
   355→                placeholder="Search products..."
   356→                value={searchQuery}
   357→                onChange={e => setSearchQuery(e.target.value)}
   358→                className="flex-1 px-4 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none"
   359→              />
   360→              <button
   361→                onClick={() => { setEditingProduct(null); setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' }); setTab('add') }}
   362→                className="bg-[#1a1a2e] text-white px-4 py-2 rounded-lg text-sm font-semibold hover:bg-[#0f3460] transition"
   363→              >
   364→                + Add Product
   365→              </button>
   366→            </div>
   367→
   368→            {loading ? (
   369→              <div className="text-center py-12 text-gray-400">Loading...</div>
   370→            ) : (
   371→              <div className="bg-white rounded-xl shadow overflow-hidden">
   372→                <table className="w-full text-sm">
   373→                  <thead className="bg-gray-50 border-b">
   374→                    <tr>
   375→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Product</th>
   376→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Brand</th>
   377→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Category</th>
   378→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Price</th>
   379→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Images</th>
   380→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Active</th>
   381→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Featured</th>
   382→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Actions</th>
   383→                    </tr>
   384→                  </thead>
   385→                  <tbody className="divide-y">
   386→                    {filtered.map(p => (
   387→                      <tr key={p.id} className="hover:bg-gray-50 transition">
   388→                        <td className="px-4 py-3">
   389→                          <div className="flex items-center gap-3">
   390→                            {p.images[0] && (
   391→                              <img src={p.images[0].url} alt="" className="w-10 h-10 object-contain rounded border bg-gray-50" onError={e => { (e.target as HTMLImageElement).style.display = 'none' }} />
   392→                            )}
   393→                            <div>
   394→                              <p className="font-medium text-gray-900">{p.name}</p>
   395→                              <p className="text-xs text-gray-400 font-mono">{p.model}</p>
   396→                            </div>
   397→                          </div>
   398→                        </td>
   399→                        <td className="px-4 py-3 text-gray-700">{p.brand.name}</td>
   400→                        <td className="px-4 py-3 text-gray-700">{p.category.name}</td>
   401→                        <td className="px-4 py-3 font-semibold text-green-600">{fmtPrice(p.price)}</td>
   402→                        <td className="px-4 py-3 text-center text-gray-500">{p.images.length}</td>
   403→                        <td className="px-4 py-3 text-center">
   404→                          <button onClick={() => toggleActive(p)} className={`px-2 py-1 rounded-full text-xs font-bold ${p.isActive ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'}`}>
   405→                            {p.isActive ? 'Live' : 'Hidden'}
   406→                          </button>
   407→                        </td>
   408→                        <td className="px-4 py-3 text-center">
   409→                          <button onClick={() => toggleFeatured(p)} className={`text-lg ${p.featured ? 'text-yellow-500' : 'text-gray-300'}`}>
   410→                            ★
   411→                          </button>
   412→                        </td>
   413→                        <td className="px-4 py-3 text-center">
   414→                          <div className="flex gap-1 justify-center">
   415→                            <button onClick={() => startEdit(p)} className="px-2 py-1 bg-blue-50 text-blue-600 rounded text-xs font-semibold hover:bg-blue-100 transition">
   416→                              Edit
   417→                            </button>
   418→                            <button onClick={() => deleteProduct(p.id)} className="px-2 py-1 bg-red-50 text-red-600 rounded text-xs font-semibold hover:bg-red-100 transition">
   419→                              Delete
   420→                            </button>
   421→                          </div>
   422→                        </td>
   423→                      </tr>
   424→                    ))}
   425→                  </tbody>
   426→                </table>
   427→                {filtered.length === 0 && (
   428→                  <div className="text-center py-12 text-gray-400">No products found</div>
   429→                )}
   430→              </div>
   431→            )}
   432→          </div>
   433→        )}
   434→
   435→        {/* ─── ADD / EDIT PRODUCT TAB ─── */}
   436→        {tab === 'add' && (
   437→          <div className="max-w-2xl mx-auto bg-white rounded-xl shadow p-6">
   438→            <h2 className="text-lg font-bold mb-4">{editingProduct ? 'Edit Product' : 'Add New Product'}</h2>
   439→            <form onSubmit={editingProduct ? handleUpdateProduct : handleAddProduct} className="space-y-4">
   440→              <div className="grid grid-cols-2 gap-4">
   441→                <div>
   442→                  <label className="block text-sm font-medium text-gray-700 mb-1">Product Name *</label>
   443→                  <input type="text" value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   444→                </div>
   445→                <div>
   446→                  <label className="block text-sm font-medium text-gray-700 mb-1">Model/SKU *</label>
   447→                  <input type="text" value={form.model} onChange={e => setForm({ ...form, model: e.target.value })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   448→                </div>
   449→              </div>
   450→
   451→              <div className="grid grid-cols-2 gap-4">
   452→                <div>
   453→                  <label className="block text-sm font-medium text-gray-700 mb-1">Price (JMD) *</label>
   454→                  <input type="number" step="0.01" value={form.price} onChange={e => setForm({ ...form, price: parseFloat(e.target.value) || 0 })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   455→                </div>
   456→                <div>
   457→                  <label className="block text-sm font-medium text-gray-700 mb-1">Size (laptop fit)</label>
   458→                  <input type="text" value={form.size} onChange={e => setForm({ ...form, size: e.target.value })} placeholder='e.g. "15.6 inch" or "N/A"' className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   459→                </div>
   460→              </div>
   461→
   462→              <div className="grid grid-cols-2 gap-4">
   463→                <div>
   464→                  <label className="block text-sm font-medium text-gray-700 mb-1">Category *</label>
   465→                  <select value={form.categoryId} onChange={e => setForm({ ...form, categoryId: e.target.value })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none">
   466→                    <option value="">Select category</option>
