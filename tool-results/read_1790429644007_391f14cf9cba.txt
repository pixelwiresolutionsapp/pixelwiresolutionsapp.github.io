     1→     1→'use client'
     2→     2→
     3→     3→import { useEffect, useState, useCallback } from 'react'
     4→     4→import { toast } from 'sonner'
     5→     5→
     6→     6→// ─── Types ───
     7→     7→interface ProductImage { id: string; url: string; alt?: string; sortOrder: number; isPrimary: boolean }
     8→     8→interface Brand { id: string; name: string; slug: string }
     9→     9→interface Category { id: string; name: string; slug: string; icon?: string; color?: string }
    10→    10→interface Product {
    11→    11→  id: string; name: string; slug: string; model: string; price: number
    12→    12→  description?: string; size?: string; color: string; isActive: boolean; featured: boolean; sortOrder: number
    13→    13→  images: ProductImage[]; brand: Brand; category: Category
    14→    14→}
    15→    15→
    16→    16→type Tab = 'products' | 'add' | 'categories' | 'brands' | 'orders'
    17→    17→
    18→    18→// ─── Component ───
    19→    19→export default function AdminDashboard() {
    20→    20→  const [tab, setTab] = useState<Tab>('products')
    21→    21→  const [products, setProducts] = useState<Product[]>([])
    22→    22→  const [categories, setCategories] = useState<Category[]>([])
    23→    23→  const [brands, setBrands] = useState<Brand[]>([])
    24→    24→  const [orders, setOrders] = useState<any[]>([])
    25→    25→  const [loading, setLoading] = useState(true)
    26→    26→  const [searchQuery, setSearchQuery] = useState('')
    27→    27→
    28→    28→  // Edit state
    29→    29→  const [editingProduct, setEditingProduct] = useState<Product | null>(null)
    30→    30→
    31→    31→  // Add product form
    32→    32→  const [form, setForm] = useState({
    33→    33→    name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827',
    34→    34→    categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0,
    35→    35→    imageUrls: ''  // comma-separated
    36→    36→  })
    37→    37→
    38→    38→  // Add category form
    39→    39→  const [catForm, setCatForm] = useState({ name: '', slug: '', icon: '', color: '#1a1a2e' })
    40→    40→  // Add brand form
    41→    41→  const [brandForm, setBrandForm] = useState({ name: '', slug: '', logoUrl: '' })
    42→    42→
    43→    43→  const API = '/api'
    44→    44→
    45→    45→  // ─── Load data ───
    46→    46→  const loadData = useCallback(async () => {
    47→    47→    try {
    48→    48→      const [prodRes, catRes, brandRes] = await Promise.all([
    49→    49→        fetch(`${API}/products?limit=500`),
    50→    50→        fetch(`${API}/categories`),
    51→    51→        fetch(`${API}/brands`),
    52→    52→      ])
    53→    53→      const prodData = await prodRes.json()
    54→    54→      const catData = await catRes.json()
    55→    55→      const brandData = await brandRes.json()
    56→    56→      setProducts(prodData.products || [])
    57→    57→      setCategories(catData || [])
    58→    58→      setBrands(brandData || [])
    59→    59→    } catch {
    60→    60→      toast.error('Failed to load data')
    61→    61→    } finally {
    62→    62→      setLoading(false)
    63→    63→    }
    64→    64→  }, [])
    65→    65→
    66→    66→  const loadOrders = useCallback(async () => {
    67→    67→    try {
    68→    68→      const res = await fetch(`${API}/orders?limit=100`)
    69→    69→      const data = await res.json()
    70→    70→      setOrders(data.orders || [])
    71→    71→    } catch {
    72→    72→      toast.error('Failed to load orders')
    73→    73→    }
    74→    74→  }, [])
    75→    75→
    76→    76→  useEffect(() => { loadData() }, [loadData])
    77→    77→  useEffect(() => { if (tab === 'orders') loadOrders() }, [tab, loadOrders])
    78→    78→
    79→    79→  // ─── Product CRUD ───
    80→    80→  const deleteProduct = async (id: string) => {
    81→    81→    if (!confirm('Delete this product? This cannot be undone.')) return
    82→    82→    try {
    83→    83→      await fetch(`${API}/products/${id}`, { method: 'DELETE' })
    84→    84→      toast.success('Product deleted')
    85→    85→      loadData()
    86→    86→    } catch {
    87→    87→      toast.error('Delete failed')
    88→    88→    }
    89→    89→  }
    90→    90→
    91→    91→  const toggleActive = async (product: Product) => {
    92→    92→    try {
    93→    93→      await fetch(`${API}/products/${product.id}`, {
    94→    94→        method: 'PUT',
    95→    95→        headers: { 'Content-Type': 'application/json' },
    96→    96→        body: JSON.stringify({ ...product, isActive: !product.isActive, brandId: product.brand.id, categoryId: product.category.id }),
    97→    97→      })
    98→    98→      toast.success(product.isActive ? 'Product hidden' : 'Product visible')
    99→    99→      loadData()
   100→   100→    } catch {
   101→   101→      toast.error('Update failed')
   102→   102→    }
   103→   103→  }
   104→   104→
   105→   105→  const toggleFeatured = async (product: Product) => {
   106→   106→    try {
   107→   107→      await fetch(`${API}/products/${product.id}`, {
   108→   108→        method: 'PUT',
   109→   109→        headers: { 'Content-Type': 'application/json' },
   110→   110→        body: JSON.stringify({ ...product, featured: !product.featured, brandId: product.brand.id, categoryId: product.category.id }),
   111→   111→      })
   112→   112→      toast.success(product.featured ? 'Unfeatured' : 'Featured!')
   113→   113→      loadData()
   114→   114→    } catch {
   115→   115→      toast.error('Update failed')
   116→   116→    }
   117→   117→  }
   118→   118→
   119→   119→  // ─── Add product ───
   120→   120→  const handleAddProduct = async (e: React.FormEvent) => {
   121→   121→    e.preventDefault()
   122→   122→    if (!form.name || !form.model || !form.categoryId || !form.brandId) {
   123→   123→      toast.error('Fill in all required fields')
   124→   124→      return
   125→   125→    }
   126→   126→    try {
   127→   127→      const images = form.imageUrls.split(',').map(u => u.trim()).filter(Boolean)
   128→   128→      const res = await fetch(`${API}/products`, {
   129→   129→        method: 'POST',
   130→   130→        headers: { 'Content-Type': 'application/json' },
   131→   131→        body: JSON.stringify({ ...form, images, price: Number(form.price), sortOrder: Number(form.sortOrder) }),
   132→   132→      })
   133→   133→      if (!res.ok) throw new Error()
   134→   134→      toast.success('Product added!')
   135→   135→      setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' })
   136→   136→      loadData()
   137→   137→      setTab('products')
   138→   138→    } catch {
   139→   139→      toast.error('Failed to add product')
   140→   140→    }
   141→   141→  }
   142→   142→
   143→   143→  // ─── Update product ───
   144→   144→  const handleUpdateProduct = async (e: React.FormEvent) => {
   145→   145→    e.preventDefault()
   146→   146→    if (!editingProduct) return
   147→   147→    try {
   148→   148→      const images = form.imageUrls.split(',').map(u => u.trim()).filter(Boolean)
   149→   149→      await fetch(`${API}/products/${editingProduct.id}`, {
   150→   150→        method: 'PUT',
   151→   151→        headers: { 'Content-Type': 'application/json' },
   152→   152→        body: JSON.stringify({
   153→   153→          name: form.name, model: form.model, price: Number(form.price),
   154→   154→          description: form.description, size: form.size, color: form.color,
   155→   155→          categoryId: form.categoryId, brandId: form.brandId,
   156→   156→          featured: form.featured, isActive: form.isActive, sortOrder: Number(form.sortOrder),
   157→   157→          images
   158→   158→        }),
   159→   159→      })
   160→   160→      toast.success('Product updated!')
   161→   161→      setEditingProduct(null)
   162→   162→      setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' })
   163→   163→      loadData()
   164→   164→      setTab('products')
   165→   165→    } catch {
   166→   166→      toast.error('Failed to update product')
   167→   167→    }
   168→   168→  }
   169→   169→
   170→   170→  const startEdit = (p: Product) => {
   171→   171→    setEditingProduct(p)
   172→   172→    setForm({
   173→   173→      name: p.name, model: p.model, price: p.price, description: p.description || '',
   174→   174→      size: p.size || 'N/A', color: p.color, categoryId: p.category.id, brandId: p.brand.id,
   175→   175→      featured: p.featured, isActive: p.isActive, sortOrder: p.sortOrder,
   176→   176→      imageUrls: p.images.map(i => i.url).join(', ')
   177→   177→    })
   178→   178→    setTab('add')
   179→   179→  }
   180→   180→
   181→   181→  // ─── Add category ───
   182→   182→  const handleAddCategory = async (e: React.FormEvent) => {
   183→   183→    e.preventDefault()
   184→   184→    if (!catForm.name) return
   185→   185→    try {
   186→   186→      const slug = catForm.slug || catForm.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   187→   187→      const res = await fetch(`${API}/categories`, {
   188→   188→        method: 'POST',
   189→   189→        headers: { 'Content-Type': 'application/json' },
   190→   190→        body: JSON.stringify({ ...catForm, slug }),
   191→   191→      })
   192→   192→      if (!res.ok) throw new Error()
   193→   193→      toast.success('Category added!')
   194→   194→      setCatForm({ name: '', slug: '', icon: '', color: '#1a1a2e' })
   195→   195→      loadData()
   196→   196→    } catch {
   197→   197→      toast.error('Failed to add category')
   198→   198→    }
   199→   199→  }
   200→   200→
   201→   201→  // ─── Add brand ───
   202→   202→  const handleAddBrand = async (e: React.FormEvent) => {
   203→   203→    e.preventDefault()
   204→   204→    if (!brandForm.name) return
   205→   205→    try {
   206→   206→      const slug = brandForm.slug || brandForm.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   207→   207→      const res = await fetch(`${API}/brands`, {
   208→   208→        method: 'POST',
   209→   209→        headers: { 'Content-Type': 'application/json' },
   210→   210→        body: JSON.stringify({ ...brandForm, slug }),
   211→   211→      })
   212→   212→      if (!res.ok) throw new Error()
   213→   213→      toast.success('Brand added!')
   214→   214→      setBrandForm({ name: '', slug: '', logoUrl: '' })
   215→   215→      loadData()
   216→   216→    } catch {
   217→   217→      toast.error('Failed to add brand')
   218→   218→    }
   219→   219→  }
   220→   220→
   221→   221→  // ─── Delete category ───
   222→   222→  const deleteCategory = async (id: string) => {
   223→   223→    if (!confirm('Delete this category? All products in it will also be deleted!')) return
   224→   224→    try {
   225→   225→      await fetch(`${API}/categories/${id}`, { method: 'DELETE' })
   226→   226→      toast.success('Category deleted')
   227→   227→      loadData()
   228→   228→    } catch { toast.error('Delete failed') }
   229→   229→  }
   230→   230→
   231→   231→  // ─── Delete brand ───
   232→   232→  const deleteBrand = async (id: string) => {
   233→   233→    if (!confirm('Delete this brand? All products under it will also be deleted!')) return
   234→   234→    try {
   235→   235→      await fetch(`${API}/brands/${id}`, { method: 'DELETE' })
   236→   236→      toast.success('Brand deleted')
   237→   237→      loadData()
   238→   238→    } catch { toast.error('Delete failed') }
   239→   239→  }
   240→   240→
   241→   241→  const fmtPrice = (p: number) => `$${p.toLocaleString('en-JM', { minimumFractionDigits: 2 })}`
   242→   242→
   243→   243→  const filtered = products.filter(p =>
   244→   244→    !searchQuery || p.name.toLowerCase().includes(searchQuery.toLowerCase()) || p.model.toLowerCase().includes(searchQuery.toLowerCase())
   245→   245→  )
   246→   246→
   247→   247→  // ─── Render ───
   248→   248→  return (
   249→   249→    <div className="min-h-screen bg-gray-50">
   250→   250→      {/* Header */}
   251→   251→      <header className="bg-[#1a1a2e] text-white shadow-lg">
   252→   252→        <div className="max-w-7xl mx-auto px-4 py-4 flex items-center justify-between">
   253→   253→          <div>
   254→   254→            <h1 className="text-xl font-bold">PixelWire Admin</h1>
   255→   255→            <p className="text-xs opacity-60">Product &amp; Inventory Management</p>
   256→   256→          </div>
   257→   257→          <a href="/" className="text-sm bg-white/10 hover:bg-white/20 px-3 py-1.5 rounded-lg transition">
   258→   258→            View Storefront
   259→   259→          </a>
   260→   260→        </div>
   261→   261→      </header>
   262→   262→
   263→   263→      {/* Tabs */}
   264→   264→      <nav className="bg-white border-b shadow-sm sticky top-0 z-40">
   265→   265→        <div className="max-w-7xl mx-auto px-4 flex gap-1 overflow-x-auto">
   266→   266→          {([
   267→   267→            ['products', 'Products', `${products.length} items`],
   268→   268→            ['add', editingProduct ? 'Edit Product' : 'Add Product', ''],
   269→   269→            ['categories', 'Categories', `${categories.length}`],
   270→   270→            ['brands', 'Brands', `${brands.length}`],
   271→   271→            ['orders', 'Orders', ''],
   272→   272→          ] as [Tab, string, string][]).map(([key, label, count]) => (
   273→   273→            <button
   274→   274→              key={key}
   275→   275→              onClick={() => setTab(key)}
   276→   276→              className={`px-4 py-3 text-sm font-medium border-b-2 transition whitespace-nowrap ${
   277→   277→                tab === key
   278→   278→                  ? 'border-[#1a1a2e] text-[#1a1a2e]'
   279→   279→                  : 'border-transparent text-gray-500 hover:text-gray-700'
   280→   280→              }`}
   281→   281→            >
   282→   282→              {label}{count ? ` (${count})` : ''}
   283→   283→            </button>
   284→   284→          ))}
   285→   285→        </div>
   286→   286→      </nav>
   287→   287→
   288→   288→      <main className="max-w-7xl mx-auto px-4 py-6">
   289→   289→        {/* ─── PRODUCTS TAB ─── */}
   290→   290→        {tab === 'products' && (
   291→   291→          <div>
   292→   292→            <div className="flex gap-3 items-center mb-4">
   293→   293→              <input
   294→   294→                type="text"
   295→   295→                placeholder="Search products..."
   296→   296→                value={searchQuery}
   297→   297→                onChange={e => setSearchQuery(e.target.value)}
   298→   298→                className="flex-1 px-4 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none"
   299→   299→              />
   300→   300→              <button
   301→   301→                onClick={() => { setEditingProduct(null); setForm({ name: '', model: '', price: 0, description: '', size: 'N/A', color: '#111827', categoryId: '', brandId: '', featured: false, isActive: true, sortOrder: 0, imageUrls: '' }); setTab('add') }}
   302→   302→                className="bg-[#1a1a2e] text-white px-4 py-2 rounded-lg text-sm font-semibold hover:bg-[#0f3460] transition"
   303→   303→              >
   304→   304→                + Add Product
   305→   305→              </button>
   306→   306→            </div>
   307→   307→
   308→   308→            {loading ? (
   309→   309→              <div className="text-center py-12 text-gray-400">Loading...</div>
   310→   310→            ) : (
   311→   311→              <div className="bg-white rounded-xl shadow overflow-hidden">
   312→   312→                <table className="w-full text-sm">
   313→   313→                  <thead className="bg-gray-50 border-b">
   314→   314→                    <tr>
   315→   315→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Product</th>
   316→   316→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Brand</th>
   317→   317→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Category</th>
   318→   318→                      <th className="px-4 py-3 text-left font-medium text-gray-600">Price</th>
   319→   319→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Images</th>
   320→   320→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Active</th>
   321→   321→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Featured</th>
   322→   322→                      <th className="px-4 py-3 text-center font-medium text-gray-600">Actions</th>
   323→   323→                    </tr>
   324→   324→                  </thead>
   325→   325→                  <tbody className="divide-y">
   326→   326→                    {filtered.map(p => (
   327→   327→                      <tr key={p.id} className="hover:bg-gray-50 transition">
   328→   328→                        <td className="px-4 py-3">
   329→   329→                          <div className="flex items-center gap-3">
   330→   330→                            {p.images[0] && (
   331→   331→                              <img src={p.images[0].url} alt="" className="w-10 h-10 object-contain rounded border bg-gray-50" onError={e => { (e.target as HTMLImageElement).style.display = 'none' }} />
   332→   332→                            )}
   333→   333→                            <div>
   334→   334→                              <p className="font-medium text-gray-900">{p.name}</p>
   335→   335→                              <p className="text-xs text-gray-400 font-mono">{p.model}</p>
   336→   336→                            </div>
   337→   337→                          </div>
   338→   338→                        </td>
   339→   339→                        <td className="px-4 py-3 text-gray-700">{p.brand.name}</td>
   340→   340→                        <td className="px-4 py-3 text-gray-700">{p.category.name}</td>
   341→   341→                        <td className="px-4 py-3 font-semibold text-green-600">{fmtPrice(p.price)}</td>
   342→   342→                        <td className="px-4 py-3 text-center text-gray-500">{p.images.length}</td>
   343→   343→                        <td className="px-4 py-3 text-center">
   344→   344→                          <button onClick={() => toggleActive(p)} className={`px-2 py-1 rounded-full text-xs font-bold ${p.isActive ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'}`}>
   345→   345→                            {p.isActive ? 'Live' : 'Hidden'}
   346→   346→                          </button>
   347→   347→                        </td>
   348→   348→                        <td className="px-4 py-3 text-center">
   349→   349→                          <button onClick={() => toggleFeatured(p)} className={`text-lg ${p.featured ? 'text-yellow-500' : 'text-gray-300'}`}>
   350→   350→                            ★
   351→   351→                          </button>
   352→   352→                        </td>
   353→   353→                        <td className="px-4 py-3 text-center">
   354→   354→                          <div className="flex gap-1 justify-center">
   355→   355→                            <button onClick={() => startEdit(p)} className="px-2 py-1 bg-blue-50 text-blue-600 rounded text-xs font-semibold hover:bg-blue-100 transition">
   356→   356→                              Edit
   357→   357→                            </button>
   358→   358→                            <button onClick={() => deleteProduct(p.id)} className="px-2 py-1 bg-red-50 text-red-600 rounded text-xs font-semibold hover:bg-red-100 transition">
   359→   359→                              Delete
   360→   360→                            </button>
   361→   361→                          </div>
   362→   362→                        </td>
   363→   363→                      </tr>
   364→   364→                    ))}
   365→   365→                  </tbody>
   366→   366→                </table>
   367→   367→                {filtered.length === 0 && (
   368→   368→                  <div className="text-center py-12 text-gray-400">No products found</div>
   369→   369→                )}
   370→   370→              </div>
   371→   371→            )}
   372→   372→          </div>
   373→   373→        )}
   374→   374→
   375→   375→        {/* ─── ADD / EDIT PRODUCT TAB ─── */}
   376→   376→        {tab === 'add' && (
   377→   377→          <div className="max-w-2xl mx-auto bg-white rounded-xl shadow p-6">
   378→   378→            <h2 className="text-lg font-bold mb-4">{editingProduct ? 'Edit Product' : 'Add New Product'}</h2>
   379→   379→            <form onSubmit={editingProduct ? handleUpdateProduct : handleAddProduct} className="space-y-4">
   380→   380→              <div className="grid grid-cols-2 gap-4">
   381→   381→                <div>
   382→   382→                  <label className="block text-sm font-medium text-gray-700 mb-1">Product Name *</label>
   383→   383→                  <input type="text" value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   384→   384→                </div>
   385→   385→                <div>
   386→   386→                  <label className="block text-sm font-medium text-gray-700 mb-1">Model/SKU *</label>
   387→   387→                  <input type="text" value={form.model} onChange={e => setForm({ ...form, model: e.target.value })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   388→   388→                </div>
   389→   389→              </div>
   390→   390→
   391→   391→              <div className="grid grid-cols-2 gap-4">
   392→   392→                <div>
   393→   393→                  <label className="block text-sm font-medium text-gray-700 mb-1">Price (JMD) *</label>
   394→   394→                  <input type="number" step="0.01" value={form.price} onChange={e => setForm({ ...form, price: parseFloat(e.target.value) || 0 })} required className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   395→   395→                </div>
   396→   396→                <div>
   397→   397→                  <label className="block text-sm font-medium text-gray-700 mb-1">Size (laptop fit)</label>
   398→   398→                  <input type="text" value={form.size} onChange={e => setForm({ ...form, size: e.target.value })} placeholder='e.g. "15.6 inch" or "N/A"' className="w-full px-3 py-2 border rounded-lg text-sm focus:border-[#1a1a2e] focus:outline-none" />
   399→   399→                </div>
