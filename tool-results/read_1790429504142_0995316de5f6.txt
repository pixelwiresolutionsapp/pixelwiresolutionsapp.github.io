   431→function slugify(text: string): string {
   432→  return text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   433→}

   438→  // Seed categories

   462→  const catMap = await prisma.category.findMany()
   463→  const brandMap = await prisma.brand.findMany()
   464→
   465→  for (let i = 0; i < products.length; i++) {

   468→    const brand = brandMap.find(b => b.slug === slugify(p.brand))!
   469→    const slug = slugify(p.name)
   470→
   471→    const product = await prisma.product.upsert({
   472→      where: { slug },

   484→      create: {

   494→        sortOrder: i,
   495→      },
   496→    })
